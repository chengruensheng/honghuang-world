- [ ] **Step 1: 修改看板驱动过程流接口（第219-236行）**

原代码：
```rust
pub async fn 看板驱动过程流接口(
    状态: State<数据服务状态>,

```

改为：
```rust
pub async fn 看板驱动过程流接口(
    状态: State<数据服务状态>,
    Query(游标): Query<事件游标>,
) -> Result<Sse<impl Stream<Item = Result<Event, Infallible>>>, StatusCode> {
    // 获取 SSE 并发许可；超限返回 429
    let 许可 = match 状态.sse_semaphore.clone().try_acquire_owned() {
        Ok(p) => p,
        Err(_) => return Err(StatusCode::TOO_MANY_REQUESTS),
    };
    let 台 = 状态.看板驱动台.clone();
    let mut 游标 = 游标.since.unwrap_or(0);
    let 流 = async_stream::stream! {
        // 持有许可直到流结束（客户端断开时 drop）
        let _许可 = 许可;
        loop {
            for 事件 in 台.过程事件增量(游标) {
                游标 = 事件.序号;
                let 载荷 = serde_json::to_string(&事件).unwrap_or_else(|_| "{}".into());
                yield Ok(Event::default().data(载荷));
            }
            tokio::time::sleep(Duration::from_millis(300)).await;
        }
    };
    Ok(Sse::new(流).keep_alive(KeepAlive::default()))
}
```

关键设计：`try_acquire_owned()` 非阻塞获取，失败立即返回 429。`_许可` 绑定在 stream! 内部，客户端断开时 stream drop → 许可自动释放。

### Task 5: 鉴权增强 — SSE 流纳入鉴权

- [ ] **Step 1: 修改鉴权层，对 SSE 路径也检查令牌**

将 `鉴权层` 函数（约第91-109行）中的 GET 放行逻辑改为：

```rust
async fn 鉴权层(
    State(鉴权令牌): State<Option<String>>,
    req: axum::extract::Request,
    next: axum::middleware::Next,
) -> Result<axum::response::Response, StatusCode> {
    let 方法 = req.method().clone();
    let 路径 = req.uri().path().to_string();

    // SSE 流端点即使 GET 也需要鉴权
    let 是SSE = 路径.starts_with("/api/dev/stream") || 路径 == "/api/dev/chat/stream";

    if 方法 == axum::http::Method::GET && !是SSE {
        return Ok(next.run(req).await);
    }

    // 未配置令牌时：仅 localhost 绑定才放行（已有启动约束兜底）
    match &鉴权令牌 {
        Some(令牌) => {
            let auth = req.headers().get("Authorization")
                .and_then(|v| v.to_str().ok())
                .unwrap_or("");
            if auth != &format!("Bearer {令牌}") {
                return Err(StatusCode::UNAUTHORIZED);
            }
        }
        None => {
            // 未配置令牌且非 localhost 的场景已被启动约束拦截
            // localhost 下放行（兼容开发环境）
        }
    }
    Ok(next.run(req).await)
}
```

### Task 6: 文件路径 canonicalize 验证

**Files:**
- Modify: `鸿蒙/基础能力-域/数据服务-府/观测读取-殿/文件查询-阁/文件-处理-园/文件处理.rs`

- [ ] **Step 1: 增强文件内容接口的路径验证（第124-145行）**

在现有的 `..` 段检测和扩展名白名单之后、`read_to_string` 之前，添加 canonicalize 验证：

```rust
pub async fn 文件内容接口(
    状态: State<数据服务状态>,
    Query(请求): Query<内容请求>,
) -> Result<Json<内容响应>, StatusCode> {
    let 相对 = 请求.路径.trim().trim_start_matches(['/', '\\']);
    // 防目录穿越：拒绝含 .. 的路径
    if 相对.is_empty() || 相对.split(['/', '\\']).any(|段| 段 == "..") {
        return Err(StatusCode::BAD_REQUEST);
    }
    // 仅文档类扩展名可读
    let 扩 = 相对.rsplit('.').next().map(|s| s.to_ascii_lowercase()).unwrap_or_default();
    if !可读扩展名.contains(&扩.as_str()) {
        return Err(StatusCode::BAD_REQUEST);
    }
    let 根 = 状态.扫描根.lock().expect("扫描根锁中毒");
    let 全路径 = std::path::Path::new(根.as_str()).join(相对);

    // ★ 新增：canonicalize 验证最终路径仍在扫描根内（防符号链接逃逸）
    let 规范路径 = 全路径.canonicalize().map_err(|_| StatusCode::NOT_FOUND)?;
    let 规范根 = std::path::Path::new(根.as_str()).canonicalize().map_err(|_| StatusCode::INTERNAL_SERVER_ERROR)?;
    if !规范路径.starts_with(&规范根) {
        tracing::warn!("文件读取路径逃逸拦截: {:?} 不在 {:?} 内", 规范路径, 规范根);
        return Err(StatusCode::FORBIDDEN);
    }

    let 内容 = std::fs::read_to_string(&规范路径).map_err(|_| StatusCode::NOT_FOUND)?;
    if 内容.len() > 内容上限 {
        return Err(StatusCode::PAYLOAD_TOO_LARGE);
    }
    Ok(Json(内容响应 { 路径: 相对.to_string(), 内容 }))
}
```

关键点：
- `canonicalize()` 会解析符号链接和 `.`/`..`，得到真实绝对路径
- `starts_with` 确保解析后的路径仍以扫描根为前缀
- 文件不存在时 `canonicalize` 返回 Err，映射为 404（与原行为一致）
- 添加 warn 日志记录逃逸尝试，便于安全审计
